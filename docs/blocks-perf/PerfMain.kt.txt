/**
 * QA ticket 98c61dd1 timing harness (not app code): the TabShotsMain window (stub daemon, temp state, 1400x860, the
 * repository plugins root) with wall-clock timings of first draw, tab switches, the New agent dialog and the role
 * manager. One JVM = one run; every line printed as "PERF <step> <ms> <semanticsNodes> <layoutNodes>".
 */
@OptIn(ExperimentalTestApi::class)
fun main(args: Array<String>) {
    //DEV: pinShotUiPrefs()
    val outFile = File(args.firstOrNull() ?: "build/perf.txt")
    val warm = System.getProperty("perf.warm")?.toIntOrNull() ?: 5
    val dir = Files.createTempDirectory("agenthq-perf").toFile()
    val root = File(dir, "plugins")
    System.getProperty(PluginFolders.ROOT_PROPERTY)?.let { File(it).copyRecursively(root) }
    val env = mapOf(PluginFolders.ROOT_ENV to root.absolutePath)
    val tLoad = System.nanoTime()
    val tabs = DesktopPlugins.load(dir, env = env::get, property = { null }).factory
    val pluginLoadMs = (System.nanoTime() - tLoad) / 1e6
    val port = java.net.ServerSocket(0).use { it.localPort }
    val provider = StubProvider(StubScript.default(), delayMs = 0)
    val n = AtomicInteger()
    val daemon = AgentHqDaemon.build(
        port = port, host = "127.0.0.1", provider = provider, runner = provider.transport()!!,
        sessionStore = JsonSessionStore(File(dir, "sessions.json")), mailStore = JsonMailStore(File(dir, "mailbox.json")),
        store = SqlAgentHqStore.open(File(dir, "agenthq.db").absolutePath), mailBaseUrl = "http://127.0.0.1:1", token = null,
        sessionIds = { "5e55a0%02d-0000-4000-8000-000000000000".format(n.incrementAndGet()) },
        //DEV: memory = object : com.superiorsoftllc.agenthq.session.MemorySource { override fun freeMb(): Long = 8192; override fun totalMb(): Long = 16384; override fun processes(): List<com.superiorsoftllc.agenthq.session.ProcMem> = emptyList() },
    )
    daemon.start()
    val scope = CoroutineScope(SupervisorJob() + Dispatchers.Default)
    val controller = Controller(scope)
    val lines = mutableListOf("PERF pluginLoad %.1f -1 -1".format(pluginLoadMs))
    runDesktopComposeUiTest(1400, 860) {
        fun sem(): Int = onAllNodes(SemanticsMatcher("any") { true }, useUnmergedTree = true).fetchSemanticsNodes().size
        fun layoutCount(): Int = runCatching {
            onAllNodes(isRoot()).fetchSemanticsNodes().sumOf { countLayout(it) }
        }.getOrDefault(-1)
        fun rec(step: String, ms: Double) {
            val l = "PERF %s %.1f %d %d".format(step, ms, sem(), layoutCount())
            println(l); lines += l
        }
        fun timed(step: String, action: () -> Unit, done: () -> Boolean) {
            val t = System.nanoTime()
            action()
            waitUntil(timeoutMillis = 30_000) { done() }
            waitForIdle()
            val ms = (System.nanoTime() - t) / 1e6
            // the draw: the topmost root rendered to an image (same readback cost on both sides)
            val tc = System.nanoTime()
            val roots = onAllNodes(isRoot())
            roots[roots.fetchSemanticsNodes().size - 1].captureToImage()
            val cap = (System.nanoTime() - tc) / 1e6
            rec(step, ms)
            val l = "PERF $step.capture %.1f -1 -1".format(cap); println(l); lines += l
        }
        fun has(tag: String) = onAllNodesWithTag(tag, useUnmergedTree = true).fetchSemanticsNodes().isNotEmpty()

        // 1. first draw: setContent until the window is idle (first composition, layout, frame)
        timed("firstDraw", { setContent { App(controller, "ws://127.0.0.1:$port/ws", null, tabs = tabs) } }, { true })
        timed("live", {}, { controller.state.value.connection.isLive })
        scope.launch { controller.spawn("say hello") }
        waitUntil(timeoutMillis = 20_000) { controller.state.value.agents.isNotEmpty() }
        waitForIdle(); Thread.sleep(1500); waitForIdle()
        rec("seeded", 0.0)

        // 2. tab switches chat (center-tab-OUTPUT) <-> git <-> board, then the New agent dialog; first pass is cold
        for (i in 0..warm) {
            val p = if (i == 0) "cold" else "warm"
            timed("$p.chatToGit", { onAllNodesWithTag("center-tab-GIT")[0].performClick() }, { true })
            timed("$p.gitToChat", { onAllNodesWithTag("center-tab-OUTPUT")[0].performClick() }, { true })
            timed("$p.boardOpen", { onNodeWithTag("sidebar-board-button").performClick() }, { has("board-modal") })
            timed("$p.boardClose", { onAllNodesWithTag("board-modal-close", useUnmergedTree = true)[0].performClick() }, { !has("board-modal") })
            timed("$p.newAgentOpen", { onNodeWithTag("sidebar-new-agent").performClick() }, { has("new-agent-dialog") })
            timed("$p.newAgentClose", {
                val inDialog = onAllNodes(hasText("Cancel") and hasAnyAncestor(hasTestTag("new-agent-dialog")))
                if (inDialog.fetchSemanticsNodes().isNotEmpty()) inDialog[0].performClick()
                else onAllNodesWithText("Cancel").let { it[it.fetchSemanticsNodes().size - 1].performClick() }
            }, { !has("new-agent-dialog") })
        }
        // 3. role manager: Configuration > Agents > Manage roles (once per JVM)
        waitUntil(timeoutMillis = 20_000) { controller.state.value.roles.isNotEmpty() }
        timed("configOpen", { onNodeWithContentDescription("Configuration").performClick() }, { has("config-tab-AGENTS") })
        timed("configAgentsTab", { onNodeWithTag("config-tab-AGENTS").performClick() }, { has("role-manager-open") })
        timed("roleManagerOpen", { onNodeWithTag("role-manager-open").performScrollTo().performClick() }, { has("role-manager-new") })
    }
    outFile.parentFile?.mkdirs()
    outFile.appendText(lines.joinToString("\n", postfix = "\n"))
    daemon.close()
    exitProcess(0)
}

/** Layout nodes under a root, through the internal LayoutNode children (reflection; -1 when it is not reachable). */
private fun countLayout(root: SemanticsNode): Int {
    val top = root.layoutInfo
    val getter = top.javaClass.methods.firstOrNull { it.name.startsWith("getChildren") && it.parameterCount == 0 && List::class.java.isAssignableFrom(it.returnType) }
        ?: return -1
    var c = 0
    val stack = ArrayDeque<Any>().apply { add(top) }
    while (stack.isNotEmpty()) {
        val x = stack.removeLast(); c++
        (getter.invoke(x) as List<*>).forEach { if (it != null) stack.add(it) }
    }
    return c
}
